## Plan de construcción del laboratorio

### Fase 1: Cliente MCP (1 semana)

```python
# tritos_lab/mcp_client.py
import subprocess, json

class MCPClient:
    def __init__(self, server="~/octave-mcp/server.py"):
        self.server = server

    def call(self, tool, params):
        req = {"jsonrpc":"2.0","id":1,"method":"tools/call",
               "params":{"name":tool,"arguments":params}}
        p = subprocess.run(
            ["python3", self.server],
            input=json.dumps(req)+"\n",
            capture_output=True, text=True, timeout=60
        )
        return json.loads(p.stdout.split("\n")[0])
```

### Fase 2: Motor de experimentación (2 semanas)

```python
# tritos_lab/experiment.py
class Experiment:
    def formula(self, expr, sistemas=None):
        """Evalúa una fórmula en múltiples sistemas"""
        results = {}
        for sys in sistemas or ["binario", "ternario", "maya", "babilonio"]:
            results[sys] = self.convert(expr, sys)
        return results

    def codigo(self, nombre, valor):
        """Prueba un código ternario"""
        if nombre == "trits5":
            return self.mcp.call("ternary_representation_tool",
                                 {"mode":"to_balanced","value":valor})
        elif nombre == "quipu":
            return self.mcp.call("ancient_calculators_tool",
                                 {"mode":"quipu","value":valor})

    def ancestro(self, civilizacion, fecha):
        """Convierte fecha al calendario ancestral"""
        return self.mcp.call("ethnomath_tool",
                             {"mode":civilizacion,"date":fecha})
```
